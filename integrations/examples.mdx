## Quickstart (Express)

From [express/quickstart.js](https://github.com/usebruno/bruno-api-docs/blob/main/examples/integrations/nodejs/express/quickstart.js):

```js
const express = require('express')
const { apiDocs } = require('@usebruno/api-docs-express')

const app = express()
const port = process.env.PORT ?? 3000

app.get('/health', (req, res) => res.json({ ok: true }))

// Path is relative to this file, not to where node was started
app.use('/docs', apiDocs({ collectionPath: './api-collection' }))

app.listen(port, () => console.log(`docs at http://localhost:${port}/docs/`))
```

## Hosted YAML over HTTPS

Point `url` at a public `https://` YAML file when the collection is already hosted:

```js
app.use('/docs', apiDocs({
  url: 'https://cdn.example.com/docs/opencollection.yml',
  environments: '*'
}))
```

```html
<script>
  new Bruno.apiDocs('#bruno-api-docs', {
    url: 'https://cdn.example.com/docs/opencollection.yml',
    theme: 'dark'
  })
</script>
```
